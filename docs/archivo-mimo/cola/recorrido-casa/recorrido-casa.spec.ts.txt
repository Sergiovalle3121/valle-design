async function anotarAvisos(page: Page) {
  await page.evaluate(() => {
    const w = window as unknown as { __avisos?: string[] };
    if (w.__avisos) return;
    w.__avisos = [];
    const recoger = () => {
      document.querySelectorAll('[data-testid="app-toast"]').forEach((n) => {
        const t = (n as HTMLElement).innerText.trim();
        if (t && !w.__avisos!.includes(t)) w.__avisos!.push(t);
      });
    };
    new MutationObserver(recoger).observe(document.body, {
      childList: true,
      subtree: true,
      characterData: true,
    });
    recoger();
  });
}

async function avisos(page: Page): Promise<string[]> {
  return page
    .evaluate(() => (window as unknown as { __avisos?: string[] }).__avisos ?? [])
    .catch(() => []);
}

async function teclear(page: Page, valor: string) {
  const input = page.getByTestId("cad-command-input");
  await input.click({ timeout: 10_000 });
  await input.fill(valor);
  await input.press("Enter");
}

async function prompt(page: Page): Promise<string> {
  const p = page.getByTestId("cad-command-prompt");
  if (!(await p.isVisible().catch(() => false))) return "(sin prompt)";
  return (await p.innerText().catch(() => "")).trim().replace(/\s+/g, " ");
}

async function log(page: Page, ultimas = 8): Promise<string> {
  const l = page.getByTestId("cad-command-line-log");
  const texto = await l.innerText({ timeout: 2_000 }).catch(() => "");
  return texto.trim().split("\n").slice(-ultimas).join(" ⏎ ");
}

async function nativos(page: Page): Promise<string> {

  const cerrarPk = page.getByLabel("Cerrar paquete de entrega");
  if (await cerrarPk.count()) await cerrarPk.first().click().catch(() => {});

async function paso(
  page: Page,
  nombre: string,
  fn: () => Promise<{ resultado: Resultado; detalle: string; evidencia?: string[] }>,
) {
  const t0 = Date.now();
  let registro: Paso;
  try {
    const r = await fn();
    registro = {
      paso: nombre,
      resultado: r.resultado,
      detalle: r.detalle,
      evidencia: r.evidencia ?? [],
      ms: Date.now() - t0,
    };
  } catch (error) {
    const mensaje = String((error as Error)?.message ?? error)
      .replace(/\u001b\[[0-9;]*m/g, "")
      .slice(0, 900);
    registro = {
      paso: nombre,
      resultado: "roto",
      detalle: `EXCEPCIÓN: ${mensaje}`,
      evidencia: [
        `prompt=${await prompt(page)}`,
        `log=${await log(page)}`,
        `nativos=${await nativos(page)}`,
        `avisos=${JSON.stringify(await avisos(page))}`,
      ],
      ms: Date.now() - t0,
    };
    // Deja la línea de comandos limpia para el paso siguiente.
    await page.keyboard.press("Escape").catch(() => {});
    await page.keyboard.press("Escape").catch(() => {});
  }
  registro.evidencia.push(`captura=${await captura(page, nombre.replace(/[^a-z0-9]+/gi, "-").slice(0, 40))}`);
  pasos.push(registro);
  console.log(`[recorrido] ${registro.resultado.toUpperCase()} · ${nombre} · ${registro.detalle}`);
  await writeFile(`${SCRATCH}/pasos.json`, JSON.stringify(pasos, null, 2));
}

// ── PDF ────────────────────────────────────────────────────────────────────
const PT_POR_MM = 72 / 25.4;
function flujosDelPdf(pdf: Buffer): string[] {
  const crudo = pdf.toString("latin1");
  const fuera: string[] = [];
  let cursor = 0;
  for (;;) {
    const inicio = crudo.indexOf("stream", cursor);
    if (inicio < 0) break;
    if (crudo.slice(inicio - 3, inicio) === "end") {
      cursor = inicio + 6;
      continue;
    }
    let datos = inicio + "stream".length;
    if (crudo[datos] === "\r") datos += 1;
    if (crudo[datos] === "\n") datos += 1;
    const fin = crudo.indexOf("endstream", datos);
    if (fin < 0) break;
    let ultimo = fin;
    while (ultimo > datos && (crudo[ultimo - 1] === "\n" || crudo[ultimo - 1] === "\r")) ultimo -= 1;
    const trozo = pdf.subarray(datos, ultimo);
    try {
      fuera.push(inflateSync(trozo).toString("latin1"));
    } catch {
      fuera.push(trozo.toString("latin1"));
    }
    cursor = fin + "endstream".length;
  }
  return fuera;
}
let textosPdf: string[] = [];
/** Cada texto del PDF con su cuerpo: `Tf` en puntos y su equivalente en mm. */
function tamanosDeTexto(pdf: Buffer): string[] {
  const t = flujosDelPdf(pdf).join("\n");
  const fuera: string[] = [];
  const re = /([\d.]+)\s+Tf[\s\S]{0,220}?\(([^)]*)\)\s*Tj/g;
  for (const m of t.matchAll(re))
    fuera.push(`${m[2]} @ ${Number(m[1]).toFixed(2)}pt=${((Number(m[1]) * 25.4) / 72).toFixed(2)}mm`);
  return fuera;
}
function tramosMm(pdf: Buffer): number[] {
  const tramos: number[] = [];
  const op = /(-?\d+(?:\.\d+)?)\s+(-?\d+(?:\.\d+)?)\s+(m|l)(?![A-Za-z0-9])/g;
  for (const f of flujosDelPdf(pdf)) {
    let actual: { x: number; y: number } | null = null;
    for (const e of f.matchAll(op)) {
      const p = { x: Number(e[1]), y: Number(e[2]) };
      if (e[3] === "l" && actual) tramos.push(Math.hypot(p.x - actual.x, p.y - actual.y) / PT_POR_MM);
      actual = p;
    }
  }
  return tramos;
}

// ── DXF ────────────────────────────────────────────────────────────────────
function dxfPares(texto: string): Array<[number, string]> {
  const lineas = texto.split(/\r?\n/);
  const pares: Array<[number, string]> = [];
  for (let i = 0; i + 1 < lineas.length; i += 2) pares.push([Number(lineas[i].trim()), lineas[i + 1].trim()]);
  return pares;
}
function dxfResumen(texto: string) {
  const pares = dxfPares(texto);
  const porTipo: Record<string, number> = {};
  const capas: Array<{ nombre: string; color: number }> = [];
  let seccion = "";
  let tabla = "";
  for (let i = 0; i < pares.length; i += 1) {
    const [c, v] = pares[i];
    if (c === 0 && v === "SECTION") {
      seccion = pares[i + 1]?.[1] ?? "";
      continue;
    }
    if (c === 0 && v === "TABLE") tabla = pares[i + 1]?.[1] ?? "";
    if (seccion === "ENTITIES" && c === 0 && v !== "ENDSEC") porTipo[v] = (porTipo[v] ?? 0) + 1;
    if (seccion === "TABLES" && tabla === "LAYER" && c === 0 && v === "LAYER") {
      let nombre = "";
      let color = 7;
      for (let j = i + 1; j < pares.length && pares[j][0] !== 0; j += 1) {
        if (pares[j][0] === 2) nombre = pares[j][1];
        if (pares[j][0] === 62) color = Number(pares[j][1]);
      }
      capas.push({ nombre, color });
    }
  }
  const version = pares.find((p, i) => p[1] === "$ACADVER" && pares[i + 1])?.[1];
  const idx = pares.findIndex((p) => p[1] === "$ACADVER");
  return { porTipo, capas, acadver: idx >= 0 ? pares[idx + 1]?.[1] : version, insunits: (() => {
    const k = pares.findIndex((p) => p[1] === "$INSUNITS");
    return k >= 0 ? pares[k + 1]?.[1] : "(no declara $INSUNITS)";
  })() };
}

function contarTipos(doc: CadDocument): Record<string, number> {
  const r: Record<string, number> = {};
  for (const e of doc.entities as Array<{ type: string }>) r[e.type] = (r[e.type] ?? 0) + 1;
  return r;
}
