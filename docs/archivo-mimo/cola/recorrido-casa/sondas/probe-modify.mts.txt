const all = new Map<string, any>(
  [...CAD_MODIFY_EDGE_COMMANDS, ...CAD_MODIFY_TRANSFORM_COMMANDS, ...CAD_MODIFY_BASIC_COMMANDS].map((c: any) => [c.name, c]),
);

const line = (id: string, x1: number, y1: number, x2: number, y2: number) => ({
  id, type: "line", start: { x: x1, y: y1, z: 0 }, end: { x: x2, y: y2, z: 0 }, layer: "0",
});

function ctx(scene: any[], selection: string[] = []) {
  const m = new Map(scene.map((e) => [e.id, e]));
  let n = 0;
  return {
    entityIds: [...m.keys()], entity: (id: string) => m.get(id), selection,
    activeLayer: "0", view: { pixelsPerUnit: 1, centerX: 0, centerY: 0 },
    newEntityId: () => `new${++n}`,
  } as any;
}
const pick = (entityId: string, x: number, y: number) => ({ kind: "entityPick", entityId, point: { x, y } });
const pt = (x: number, y: number) => ({ kind: "point", point: { x, y }, source: "pointer" });
const enter = { kind: "enter" };
const cancel = { kind: "cancel" };
const kw = (keyword: string) => ({ kind: "keyword", keyword });
const dist = (value: number) => ({ kind: "distance", value });

function drive(name: string, c: any, inputs: any[]) {
  const d = all.get(name);
  let s = d.begin(c);
  const trace: string[] = [`[begin] ${s.prompt?.message} accepts=${s.accepts} preview=${s.preview ? s.preview.length : "none"}`];
  for (const i of inputs) {
    if (s.result) break;
    s = d.step(s.state, i, c);
    trace.push(`[${i.kind}${i.keyword ? ":" + i.keyword : ""}${i.entityId ? ":" + i.entityId : ""}] -> "${s.prompt?.message}" accepts=${s.accepts} preview=${s.preview ? s.preview.length : "none"}${s.result ? " RESULT=" + s.result.kind : ""}`);
  }
  return { s, trace };
}

// 1. TRIM: linea que cruza dos muros; recortar ambos voladizos (dos clics sobre la MISMA linea)
{
  const scene = [line("h", 0, 0, 1000, 0), line("wL", 100, -100, 100, 100), line("wR", 900, -100, 900, 100)];
  const { s, trace } = drive("TRIM", ctx(scene), [enter, pick("h", 50, 0), pick("h", 950, 0), enter]);
  console.log("\n=== 1. TRIM dos voladizos de la misma linea ===");
  trace.forEach((t) => console.log(t));
  console.log(JSON.stringify(s.result, null, 1));
}
