const registry = createCadCommandRegistry([...CAD_DRAW_BASIC_COMMANDS, ...CAD_DRAW_WALL_COMMANDS]);

let nextId = 0;
function ctx(cursor?: { x: number; y: number }, vars?: any) {
  return {
    entityIds: [],
    selection: [],
    activeLayer: "0",
    view: { pixelsPerUnit: 1, centerX: 0, centerY: 0 },
    cursor,
    variables: vars,
    newEntityId: () => `e${++nextId}`,
  };
}

function drive(tokens: string[], cursor?: { x: number; y: number }, vars?: any) {
  let state: any = EMPTY_CAD_COMMAND_ENGINE;
  const effects: any[] = [];
  for (const t of tokens) {
    const action: any =
      t === "<ENTER>" ? { kind: "input", input: { kind: "enter" } } : { kind: "token", value: t };
    const r = cadCommandEngineReduce(state, action, ctx(cursor, vars), registry);
    state = r.state;
    effects.push(...r.effects);
  }
  return { state, effects };
}

const geom = (effects: any[]) =>
  effects
    .filter((e) => e.kind === "execute")
    .flatMap((e) => e.commands ?? [])
    .map((c: any) => JSON.stringify(c).slice(0, 320));
const msgs = (effects: any[]) =>
  effects.filter((e) => e.kind === "message").map((e) => `[${e.level}] ${e.text}`);

function show(title: string, r: { state: any; effects: any[] }) {
  console.log("\n### " + title);
  const g = geom(r.effects);
  if (g.length) g.forEach((x) => console.log("   GEOM " + x));
  else console.log("   GEOM (ninguna)");
  msgs(r.effects).forEach((m) => console.log("   MSG  " + m));
}
