        ForEach-Object { $_.Substring(3) } | Where-Object { $_ -match '(^|/)package(-lock)?\.json$' })

# Rebasa la rama sobre origin/main. Con $desde, solo mueve los commits posteriores a ese SHA (rebase --onto).
function Rebasar-Sobre-Main([string]$desde) {
    $antes = Head-Rama
    if ($desde) { & git -C $Repo rebase --onto origin/main $desde 2>&1 | Out-Null }
    else { & git -C $Repo rebase origin/main 2>&1 | Out-Null }
    for ($i = 0; $i -lt 200; $i++) {
        if (-not (Test-Path -LiteralPath (Join-Path $Repo '.git\rebase-merge'))) { return $true }
        $conflictos = @(& git -C $Repo diff --name-only --diff-filter=U 2>$null)
        # Dependencias: siempre la version de main, porque la mision prohibe tocarlas. En un rebase, --ours es la base nueva.
        $deps = @($conflictos | Where-Object { $_ -match '(^|/)package(-lock)?\.json$' })
        if ($deps.Count -gt 0) {
            & git -C $Repo checkout --ours -- $deps 2>&1 | Out-Null
            & git -C $Repo add -- $deps 2>&1 | Out-Null
            Log ('Rebase: conflicto de dependencias resuelto con la version de main: ' + ($deps -join ', '))
            $conflictos = @($conflictos | Where-Object { $_ -notmatch '(^|/)package(-lock)?\.json$' })
        }
        if ($conflictos.Count -eq 1 -and $conflictos[0] -eq $Registro) {
            Push-Location -LiteralPath $Repo
            & $node $Resolver 2>&1 | Out-Null
            $ok = ($LASTEXITCODE -eq 0)
            Pop-Location
            if (-not $ok) { break }
            & git -C $Repo add -- $Registro 2>&1 | Out-Null
            $conflictos = @()
        }
        if ($conflictos.Count -gt 0) {
            $script:UltimoConflicto = ($conflictos -join ', ')
            Log ('Rebase con conflictos que no resuelvo solo: ' + $script:UltimoConflicto)
            break
        }
        & git -C $Repo rebase --continue 2>&1 | Out-Null
        if ($LASTEXITCODE -ne 0 -and @(& git -C $Repo diff --name-only --diff-filter=U 2>$null).Count -eq 0) {
            & git -C $Repo rebase --skip 2>&1 | Out-Null
        }
    }
    & git -C $Repo rebase --abort 2>&1 | Out-Null
    & git -C $Repo reset -q --hard $antes 2>&1 | Out-Null
    return $false
}

# Salida cuando el rebase es imposible: una fusion normal. El rebase repite los 200+ commits uno a
# uno y choca en el que toco el fichero; la fusion compara SOLO el estado final, asi que si MiMo ya
# dejo ese fichero como main (que es lo que se le pide), entra limpia. Ademas deja la rama
# conteniendo main, con lo que la siguiente integracion ni siquiera necesita rebasar.
function Fusionar-Main {
    & git -C $Repo merge --no-edit origin/main 2>&1 | Out-Null
    if ($LASTEXITCODE -eq 0) { return $true }
    $conf = @(& git -C $Repo diff --name-only --diff-filter=U 2>$null)
    # Dependencias: siempre la version de main, que es lo que manda la mision. En una fusion, --theirs.
    $deps = @($conf | Where-Object { $_ -match '(^|/)package(-lock)?\.json$' })
    if ($deps.Count -gt 0) {
        & git -C $Repo checkout --theirs -- $deps 2>&1 | Out-Null
        & git -C $Repo add -- $deps 2>&1 | Out-Null
        $conf = @($conf | Where-Object { $_ -notmatch '(^|/)package(-lock)?\.json$' })
    }
    # El registro de desarrollo asistido choca casi siempre (las dos ramas anaden entradas) y tiene
    # resolutor propio, el mismo que usa el rebase.
    if ($conf.Count -eq 1 -and $conf[0] -eq $Registro) {
        Push-Location -LiteralPath $Repo
        & $node $Resolver 2>&1 | Out-Null
        $ok = ($LASTEXITCODE -eq 0)
        Pop-Location
        if ($ok) {
            & git -C $Repo add -- $Registro 2>&1 | Out-Null
            $conf = @()
        }
    }
    if ($conf.Count -eq 0) {
        & git -C $Repo commit --no-edit 2>&1 | Out-Null
        if ($LASTEXITCODE -eq 0) { return $true }
    }
    # NO se aborta: la fusion se deja ABIERTA con sus marcadores en el disco. La caja de MiMo bloquea
    # `git merge` (16-sep, vuelta-6: «integration is the orchestrator's job») pero NO bloquea editar,
    # `git add` ni `git commit`: dejandosela abierta, MiMo si puede terminarla. Abortar era lo que
    # borraba el trabajo de cada ciclo y hacia el bucle infinito.
    $script:ConflictoFusion = ($conf -join ', ')
    return $false
}

# El artefacto docs/cad/evidence/command-integrity.json es DETERMINISTA para un arbol dado: en
# cuanto cambia el censo de comandos, el gate del CI cae con «no coincide con lo que el arbol
# genera hoy». Paso el 17-sep: el commit que limpio 19 huerfanos dejo el artefacto en 334 cuando
# el arbol ya daba 315, y eso son 16 minutos de corrida tirados sin que nadie aprenda nada nuevo.
# Aqui se refresca y se commitea SOLO cuando el UNICO reproche es ese desajuste. Los dientes del
# gate (ROJO, no-concluyentes sin declarar, exenciones invalidas) siguen fallando y se le pasan a
# MiMo tal cual: esto no afloja nada, solo deja de tirar corridas por un fichero generado. El
# antes/despues queda en el resumen para que un desplome de «muta» no pueda pasar en silencio.
function Reparar-ArtefactoIntegridad {
    $script:SondaMalos = $null
    if (Test-Path -LiteralPath (Join-Path $Repo '.git\MERGE_HEAD')) { return }
    $art = Join-Path $Repo 'docs\cad\evidence\command-integrity.json'
    $sal = & cmd.exe /c ('cd /d ' + $Repo + ' && node scripts/cad/check-command-integrity.mjs 2>&1')
    if ($LASTEXITCODE -eq 0) { return }
    $texto = (@($sal) -join "`n")
    $reparable = ($texto -match 'command-integrity\.json no coincide') -and ($texto -notmatch 'Gate de integridad de comandos')
    if (-not $reparable) {
        $lineas = New-Object System.Collections.ArrayList
        [void]$lineas.Add('### sonda de integridad  (node scripts/cad/check-command-integrity.mjs)')
        foreach ($l in @($sal | Select-Object -Last 25)) { [void]$lineas.Add('    ' + [string]$l) }
        [void]$lineas.Add('')
        $script:SondaMalos = $lineas
        Log 'Aviso local: sonda de integridad en rojo por su CONTENIDO (no la refresco; se lo paso a MiMo).'
        return
    }
    $antes = '(ilegible)'
    try {
        $j0 = Get-Content -LiteralPath $art -Raw | ConvertFrom-Json
        $antes = [string]$j0.total + ' comandos, muta ' + $j0.verdicts.muta + ', delegado ' + $j0.verdicts.delegado
    } catch { $antes = '(ilegible)' }
    & cmd.exe /c ('cd /d ' + $Repo + ' && node scripts/cad/check-command-integrity.mjs --write 2>&1') | Out-Null
    if ($LASTEXITCODE -ne 0) { Log 'Artefacto de integridad: --write salio en rojo; lo dejo como estaba.'; return }
    $sucio = & git -C $Repo diff --name-only -- docs/cad/evidence/command-integrity.json
    if (-not $sucio) { return }
    $j = $null
    try { $j = Get-Content -LiteralPath $art -Raw | ConvertFrom-Json } catch { $j = $null }
    if (-not $j) { Log 'Artefacto de integridad: no pude leer el resultado; lo dejo como estaba.'; return }
    $v = $j.verdicts
    # OJO con el nombre: en PowerShell las variables son insensibles a mayusculas, asi que llamar
    # a esto `$resumen` sombreaba al `$Resumen` global —la RUTA del log— y la linea del refresco
    # acababa escrita en un fichero con ese texto por nombre en la raiz del repo (17-sep 19:39).
    $cifras = [string]$j.total + ' comandos, muta ' + $v.muta + ', delegado ' + $v.delegado +
              ', informa ' + $v.informa + ', honesto-limitado ' + $v.'honesto-limitado' +

# node_modules enlazados). `git clean` sin -x no toca lo ignorado (node_modules, .next).
$RepoCandados = 'D:\dev\vd-candados'
function Preparar-Candados([string]$sha) {
    if (-not (Test-Path -LiteralPath (Join-Path $RepoCandados '.git'))) {
        & git -C $Repo worktree prune 2>$null | Out-Null
        & git -C $Repo worktree add --detach $RepoCandados $sha 2>&1 | Out-Null
        if ($LASTEXITCODE -ne 0) { Log 'Candados: no pude crear el worktree de candados.'; return $false }
        foreach ($d in @('', 'apps\api', 'apps\web', 'packages\dwg-codec')) {
            $src = if ($d) { Join-Path (Join-Path $Repo $d) 'node_modules' } else { Join-Path $Repo 'node_modules' }
            $dst = if ($d) { Join-Path (Join-Path $RepoCandados $d) 'node_modules' } else { Join-Path $RepoCandados 'node_modules' }
            if ((Test-Path -LiteralPath $src) -and -not (Test-Path -LiteralPath $dst)) {
                & cmd.exe /c ('mklink /J "' + $dst + '" "' + $src + '"') | Out-Null
            }
        }
    }
    & git -C $RepoCandados checkout --detach -f $sha 2>&1 | Out-Null
    if ($LASTEXITCODE -ne 0) { Log ('Candados: no pude poner el worktree en ' + $sha + '.'); return $false }
    & git -C $RepoCandados clean -fdq 2>&1 | Out-Null
    return $true
}

function Subir([string]$sha) {

    # el lanzador se queda el mismo sin poder empujar (10 pushes con codigo 128 el 16-sep, 05:01-09:32).
    Poner-Guardias $false
    $salida = & git -C $Repo push --force-with-lease origin ($sha + ':refs/heads/' + $Rama) 2>&1
    $rc = $LASTEXITCODE
    Log ('Push de la rama (' + $sha.Substring(0, 8) + '): codigo ' + $rc)
    if ($rc -ne 0) { Log ('  ' + (($salida | Select-Object -Last 3) -join ' | ')) }
    return ($rc -eq 0)
}

# Estados: mergeada | cerrada | conflicto con main | fallo: ... | verde bloqueada | en curso
function Estado-PR([string]$url) {
    $j = $null
    try { $j = (& gh pr view $url --repo $Slug --json state,mergeable,mergeStateStatus,statusCheckRollup 2>$null | Out-String) | ConvertFrom-Json } catch { }
    if (-not $j) { return 'en curso' }
    if ($j.state -eq 'MERGED') { return 'mergeada' }
    if ($j.state -eq 'CLOSED') { return 'cerrada' }
    if ($j.mergeable -eq 'CONFLICTING') { return 'conflicto con main' }
    # Comodines sin caracteres no ASCII: la salida de gh llega con la pagina de codigos de la consola.
    $requeridos = @('*Build*Test*Lint*', 'E2E Playwright (PostgreSQL*', 'Gitleaks*')
    $rollup = @($j.statusCheckRollup)
    $verdes = $true
    foreach ($patron in $requeridos) {
        $corridas = @($rollup | Where-Object { $_.name -like $patron })
        if (@($corridas | Where-Object { $_.conclusion -eq 'FAILURE' -or $_.conclusion -eq 'TIMED_OUT' }).Count -gt 0) {
            return ('fallo: ' + $patron.Trim('*'))
        }
        if (@($corridas | Where-Object { $_.status -and $_.status -ne 'COMPLETED' }).Count -gt 0) { $verdes = $false }
        if (@($corridas | Where-Object { $_.conclusion -eq 'SUCCESS' }).Count -eq 0) { $verdes = $false }
    }
    if ($verdes -and $j.mergeStateStatus -eq 'BLOCKED') { return 'verde bloqueada' }
    return 'en curso'
}

        # Y SIEMPRE las 12 lineas anteriores a CADA «Process completed with exit code», en crudo.
        # Ahi esta siempre la causa: el filtro por palabras se dejo fuera tres veces la linea real
        # («desactualizado», «Trinquete de lint: FALLO») y la cola del log entero cae en el paso de
        # subir informes de Lighthouse, que es ruido, porque --log-failed concatena todos los pasos.
        $contexto = New-Object System.Collections.ArrayList
        for ($i = 0; $i -lt $bruto.Count; $i++) {
            if ([string]$bruto[$i] -match 'Process completed with exit code') {
                $desde = [Math]::Max(0, $i - 12)
                [void]$contexto.Add('')
                [void]$contexto.Add('--- lo que se imprimio justo antes de un paso fallido ---')
                for ($j = $desde; $j -le $i; $j++) { [void]$contexto.Add([string]$bruto[$j]) }
            }
        }
        $sel = @($errores) + @($clave | Select-Object -Last 20) + @($contexto)
        if ($sel.Count -eq 0) { $sel = @($bruto | Select-Object -Last 60) }
        $log = ($sel -join "`n")
        if ($log.Length -gt 8000) { $log = $log.Substring($log.Length - 8000) }
    }
    $pie = "`n`n## Si arriba no ves la causa`n`nA veces `gh` devuelve el log vacio porque la corrida acaba de terminar. Entonces:`n" +
           "1. Lee ``.mimocode/bloqueos-actuales.md``: lo escribe el supervisor con lo que de verdad impide el merge, medido, y NO se sobrescribe.`n" +
           "2. O saca la causa tu mismo: ``gh run list --branch " + $Rama + " --event pull_request --limit 1`` y despues ``gh run view <id> --log-failed | tail -300``.`n" +
           "3. Y corre la bateria local de ``.mimocode/cola-vallecad.md`` (T16): en segundos te dice lo mismo que el CI en 35 minutos.`n"
    $texto = "# Fallo de CI en la integracion nocturna`n`nPR: " + $url + "`nChecks en rojo: " + $motivo + "`n`n## Ultimas lineas de los jobs fallidos`n`n" + $log + $pie
    [System.IO.File]::WriteAllText($CiFallo, $texto, (New-Object System.Text.UTF8Encoding $false))
    Log ('Fallo de CI guardado en .mimocode/ci-fallo.md (' + $motivo + ').')
}

function Guardar-Conflicto([string]$ficheros) {
    $ruta = Join-Path $Repo '.mimocode\conflicto.md'
    $t = "# Tu rama choca con main (" + (Get-Date -Format 'yyyy-MM-dd HH:mm') + ")`n`n"
    $t += "El lanzador intento mover tus commits encima de main y no pudo. Ficheros en conflicto:`n`n"
    $t += "    " + $ficheros + "`n`n"
    $t += "Main avanza porque se van mergeando PRs pequenas; algunas arreglan lo mismo que tu tocaste.`n`n"
    $t += "LA FUSION YA ESTA ABIERTA EN TU ARBOL. No ejecutes git merge, git rebase ni git pull: tu entorno\`n"
    $t += "los bloquea. Resuelve los marcadores a mano, git add de cada fichero, y git commit al final.`n`n"
    $t += "Criterio: marca, rutas, assets y textos publicos -> la version de MAIN.`n"
    $t += "          logica CAD, componentes y specs -> la version de TU rama.`n"
    $t += "          package.json y package-lock.json -> siempre MAIN.`n"
    [System.IO.File]::WriteAllText($ruta, $t, (New-Object System.Text.UTF8Encoding $false))
}

function Previa-Abierta {
    if ($script:PRPrevia -le 0) { return $false }
    $estado = [string](& gh pr view $script:PRPrevia --repo $Slug --json state --jq '.state' 2>$null | Select-Object -First 1)
    if ($estado -eq 'MERGED' -or $estado -eq 'CLOSED') {
        Log ('La #' + $script:PRPrevia + ' ya no esta abierta (' + $estado + ').')
        $script:PRPrevia = 0
        return $false
    }
    return $true
}

# --- Pasos de integracion. Los que devuelven $true han pausado a MiMo: el bucle lo relanza con $script:Mensaje.

function Integrar {
    # 16-sep: el titular quiere ver el avance en main cuanto antes; 2 h entre integraciones era
    # el cuello de botella. Ahora 20 min: el lanzador rebasa, sube y actualiza la PR cada 20 min
    # si hay commits nuevos; el CI y el merge titular hacen el resto.
    $script:ProximaIntegracion = (Get-Date).AddMinutes(20)
    & git -C $Repo fetch -q --prune origin 2>&1 | Out-Null
    if (Previa-Abierta) {
        if (-not $script:AvisoPrevia) {
            Log ('Integracion: espero a que se mergee la #' + $script:PRPrevia + ', que arregla el CI de main. Lo miro cada 5 min.')
            $script:AvisoPrevia = $true
        }
        $script:ProximaIntegracion = (Get-Date).AddMinutes(5)
        return $false
    }
    # POST-SQUASH (17-sep). El repo solo permite SQUASH y borra la rama al mergear: el contenido
    # entra en main pero los commits de la rama NO quedan como ancestros suyos. Si no se hace nada,
    # la rama arrastra para siempre sus commits viejos, cada PR nueva anuncia «279 commits» y su
    # diff se calcula desde una base muerta. Cuando el arbol de la rama YA es identico al de main,
    # esos commits solo estorban: se mueve el puntero con `checkout -B`, que NO toca el arbol de
    # trabajo (un `reset --hard` le borraria a MiMo lo que tenga sin comitear).
    if ((Commits-Sobre-Main) -gt 0) {
        & git -C $Repo diff --quiet origin/main $Rama 2>&1 | Out-Null
        if ($LASTEXITCODE -eq 0) {
            $viejos = Commits-Sobre-Main
            & git -C $Repo checkout -B $Rama origin/main 2>&1 | Out-Null
            if ($LASTEXITCODE -eq 0) {
                Log ('Rama realineada con main tras el squash: ' + $viejos + ' commits viejos retirados; su contenido ya estaba dentro de main.')
            } else {
                Log 'No pude realinear la rama con main tras el squash (arbol ocupado). Lo reintento en el proximo ciclo.'
            }
        }
    }
    if ((Commits-Sobre-Main) -le 0) {
        Log 'Integracion: no hay commits nuevos sobre main.'
        $script:ProximaIntegracion = (Get-Date).AddMinutes(30)
        return $false
    }
    $deps = @(& git -C $Repo diff --name-only ('origin/main...' + $Rama) 2>$null | Where-Object { $_ -match '(^|/)package(-lock)?\.json$' })
    if ($deps.Count -gt 0) {
        Log ('Integracion bloqueada: la rama cambia dependencias (' + ($deps -join ', ') + ') y la mision lo prohibe. Pido a MiMo que lo revierta.')
        $script:MensajeAviso = $MsgDeps + $MsgContinuar
        $script:PorAvisar = $true
        $script:AvisoDesde = Get-Date
        $script:ProximaIntegracion = (Get-Date).AddMinutes(30)
        return $false
    }
    $pausado = $false
    # Si hay una fusion con main ABIERTA esperando a MiMo, no se hace nada mas: ni gates, ni push, ni
    # rebase. El arbol tiene marcadores de conflicto y cualquier comprobacion daria rojo por eso.
    if (Test-Path -LiteralPath (Join-Path $Repo '.git\MERGE_HEAD')) {
        if (-not $script:FusionAvisada) {
            Log 'Integracion: fusion con main ABIERTA en el arbol. Se la paso a MiMo y espero a que la termine.'
            $script:MensajeAviso = $MsgConflicto + $MsgContinuar
            $script:PorAvisar = $true
            $script:AvisoDesde = Get-Date
            $script:FusionAvisada = $true
            $script:ProximaIntegracion = (Get-Date).AddMinutes(15)
            return $true
        }
        Log 'Integracion: la fusion con main sigue abierta; MiMo la esta resolviendo.'
        $script:ProximaIntegracion = (Get-Date).AddMinutes(10)
        return $false
    }
    $script:FusionAvisada = $false
    if (-not (Contiene-Main $Rama)) {
        if (-not (Arbol-Limpio)) {
            if (-not $script:AvisoArbol) {
                Log 'Integracion: main avanzo; espero a que MiMo deje el arbol limpio para fusionar.'
                $script:AvisoArbol = $true
                $script:ArbolDesde = Get-Date
            }
            # Tope de 15 min, pero SOLO si MiMo esta de verdad atascado. Se mira si el conjunto de
            # ficheros sucios CAMBIA: si cambia, MiMo esta trabajando y el reloj se reinicia; si es
            # identico 15 min, esta parado y se le pausa. Sin esta distincion se le cortaba la vuelta
            # a mitad cada 25 min y perdia el trabajo sin commitear (17-sep 12:34).
            $huella = ((& git --no-optional-locks -C $Repo status --porcelain --untracked-files=all 2>$null) -join '|')
            if ($huella -ne $script:ArbolHuella) {
                $script:ArbolHuella = $huella
                $script:ArbolDesde = Get-Date
            }
            if ($script:ArbolDesde -and ((Get-Date) - $script:ArbolDesde).TotalMinutes -ge 15) {
                Log 'Arbol sucio y SIN cambios 15 min (MiMo parado): lo pauso para poder integrar.'
                $script:AvisoArbol = $false
                $script:ArbolDesde = $null
                Detener-Mimo
                if (-not $script:Mensaje) { $script:Mensaje = $MsgContinuar }
            } else {
                $script:ProximaIntegracion = (Get-Date).AddMinutes(2)
                return $false
            }
        }
        $script:AvisoArbol = $false
        Log 'Integracion: pauso a MiMo un momento para rebasar sobre main.'
        Detener-Mimo
        $pausado = $true
        if (-not $script:Mensaje) { $script:Mensaje = $MsgContinuar }
        if (-not (Arbol-Limpio)) {
            Log 'Integracion aplazada 10 min: quedaron cambios a medias al pausar.'
            $script:ProximaIntegracion = (Get-Date).AddMinutes(10)
            return $true
        }
        # 16-sep 19:45: la FUSION va primero. El rebase repite 226 commits uno a uno y choca en
        # ficheros cuyo contenido final es identico (Toggle.tsx: mismo blob en las dos ramas); la
        # fusion compara solo el estado final. Si la fusion tampoco entra, se queda ABIERTA y la
        # termina MiMo, que si puede editar, `git add` y `git commit`.
        if (Fusionar-Main) {
            Log ('Fusion con main hecha: ' + (Commits-Sobre-Main) + ' commits sobre main.')
        } elseif (Test-Path -LiteralPath (Join-Path $Repo '.git\MERGE_HEAD')) {
            $ficheros = $script:ConflictoFusion
            if (-not $ficheros) { $ficheros = 'sin detalle' }
            Log ('Integracion: la fusion con main choca en ' + (@($ficheros -split ', ').Count) + ' ficheros. La dejo ABIERTA y se la paso a MiMo.')
            Guardar-Conflicto $ficheros
            Remove-Item -LiteralPath $CiFallo -ErrorAction SilentlyContinue
            $script:MensajeAviso = $MsgConflicto + $MsgContinuar
            $script:PorAvisar = $true
            $script:AvisoDesde = Get-Date
            $script:FusionAvisada = $true
            $script:ProximaIntegracion = (Get-Date).AddMinutes(15)
            return $true
        } elseif (Rebasar-Sobre-Main '') {
            Log ('Rebase sobre main hecho: ' + (Commits-Sobre-Main) + ' commits encima.')
        } else {
            $ficheros = $script:ConflictoFusion
            if (-not $ficheros) { $ficheros = $script:UltimoConflicto }
            if (-not $ficheros) { $ficheros = 'sin detalle' }
            Log ('Integracion: la fusion con main choca en ' + (@($ficheros -split ', ').Count) + ' ficheros. La dejo ABIERTA y se la paso a MiMo.')
            Guardar-Conflicto $ficheros
            $script:MensajeAviso = $MsgConflicto + $MsgContinuar
            $script:PorAvisar = $true
            $script:AvisoDesde = Get-Date
            $script:ProximaIntegracion = (Get-Date).AddMinutes(15)
            return $true
        }
        if ((Commits-Sobre-Main) -le 0) { Log 'Integracion: tras rebasar, todo ya estaba en main.'; return $true }
    }
    Reparar-ArtefactoIntegridad
    $sha = Head-Rama
    $n = [int](& git -C $Repo rev-list --count ('origin/main..' + $sha) 2>$null | Select-Object -First 1)
    # 16-sep: CANDADO LOCAL antes de subir. MiMo declaraba resuelto con specs rojos y el CI lo
    # descubria 45 min despues. Ahora se corren aqui los checks del CI; si alguno falla NO se sube:
    # el fallo va a ci-fallo.md y se relanza a MiMo con el, igual que con un fallo real del CI.
    if (-not (Preparar-Candados $sha)) { $script:ProximaIntegracion = (Get-Date).AddMinutes(20); return $pausado }
    if (-not (Gates-Locales)) { $script:ProximaIntegracion = (Get-Date).AddMinutes(20); return $pausado }
    if (-not (Subir $sha)) { $script:ProximaIntegracion = (Get-Date).AddMinutes(30); return $pausado }
    $url = Asegurar-PR $sha $n
    if (-not $url) { $script:ProximaIntegracion = (Get-Date).AddMinutes(30); return $pausado }
    $script:PRUrl = $url
    $script:ShaSubido = $sha
    $script:PRDesde = Get-Date
    $script:ProximaRevisionCI = (Get-Date).AddMinutes(8)
    $script:VerdeVista = $false
    $script:IntentosAdmin = 0
    $script:AvisoBloqueo = $false
    Log ('CI en marcha para ' + $sha.Substring(0, 8) + ' (' + $n + ' commits). MiMo sigue trabajando mientras tanto.')
    return $pausado
}

function Revisar-PR {
    if ((Get-Date) -lt $script:ProximaRevisionCI) { return }
    $script:ProximaRevisionCI = (Get-Date).AddMinutes(4)
    $res = Estado-PR $script:PRUrl
    if ($res -eq 'en curso') {
        if (((Get-Date) - $script:PRDesde).TotalMinutes -ge 180 -and -not $script:AvisoLento) {
            Log 'El CI de la PR lleva 3 h sin veredicto; sigo esperando.'
            $script:AvisoLento = $true
        }
        # Tope duro: a las 4 h sin veredicto se deja de vigilar esa PR y se vuelve a integrar. Sin esto,
        # una corrida colgada deja al lanzador esperando para siempre y no se mergea nada mas.
        if (((Get-Date) - $script:PRDesde).TotalMinutes -ge 240) {
            Log 'El CI de la PR lleva 4 h sin veredicto: dejo de vigilarla y vuelvo a integrar sobre lo nuevo de MiMo.'
            $script:PRUrl = $null
            $script:AvisoLento = $false
            $script:VerdeVista = $false
            $script:IntentosAdmin = 0
            $script:ProximaIntegracion = Get-Date
        }
        return
    }
    if ($res -eq 'verde bloqueada') {
        # Se da una vuelta de margen al auto-merge antes de actuar.
        if (-not $script:VerdeVista) { $script:VerdeVista = $true; $script:ProximaRevisionCI = (Get-Date).AddMinutes(2); return }
        if ($script:IntentosAdmin -ge 3) {
            if (-not $script:AvisoBloqueo) { Log ('PR en verde pero bloqueada: queda para que Sergio la mergee en GitHub (' + $script:PRUrl + ').'); $script:AvisoBloqueo = $true }
            return
        }
        $script:IntentosAdmin++
        $sha8 = $script:ShaSubido.Substring(0, 8)
        if ($Merge -eq 'titular') {
            $salida = & gh pr merge $script:PRUrl --repo $Slug --squash --admin --match-head-commit $script:ShaSubido 2>&1
            $rc = $LASTEXITCODE
            Log ('Checks requeridos verdes sobre ' + $sha8 + ' pero PR bloqueada por las corridas push canceladas: merge como titular (autorizado por Sergio), codigo ' + $rc + '.')
            if ($rc -ne 0) { Log ('  ' + (($salida | Select-Object -Last 2) -join ' | ')) }
            $script:ProximaRevisionCI = (Get-Date).AddMinutes(1)
        } elseif ($Merge -eq 'repetir') {
            $idPush = [string](& gh run list --repo $Slug --workflow CI --branch $Rama --event push --commit $script:ShaSubido --limit 1 --json databaseId,conclusion --jq '.[0] | select(.conclusion == "cancelled") | .databaseId' 2>$null | Select-Object -First 1)
            $script:IntentosAdmin = 3
            if ($idPush) {
                & gh run rerun $idPush --repo $Slug 2>&1 | Out-Null
                Log ('Checks requeridos verdes sobre ' + $sha8 + ' pero PR bloqueada por la corrida push cancelada ' + $idPush + ': la relanzo para que el auto-merge actue (codigo ' + $LASTEXITCODE + ').')
                $script:ProximaRevisionCI = (Get-Date).AddMinutes(10)
            }
        } else {
            $script:IntentosAdmin = 3
            Log ('PR en verde sobre ' + $sha8 + ' y lista para mergear en GitHub: ' + $script:PRUrl)
        }
        return
    }
    Log ('Resultado de la integracion: ' + $res)
    $url = $script:PRUrl
    $script:PRUrl = $null
    $script:AvisoLento = $false
    if ($res -eq 'mergeada') { $script:PorReubicar = $true; return }
    if ($res -like 'fallo*') {
        & git -C $Repo fetch -q --prune origin 2>&1 | Out-Null
        if (-not (Contiene-Main $script:ShaSubido)) {
            Log 'El fallo es sobre una base vieja (main avanzo desde el push): reintegro sobre main sin molestar a MiMo.'
            $script:ProximaIntegracion = Get-Date
            return
        }
        # Fallo de infraestructura: con los repos privados, el CI no puede clonar el repo hermano del corpus DWG
        # (git sale con 128). No es de MiMo: no se le avisa y se reintenta cada 3 h hasta que Sergio dé acceso.
        $runId = [string](& gh run list --repo $Slug --branch $Rama --workflow CI --event pull_request --limit 1 --json databaseId --jq '.[0].databaseId' 2>$null | Select-Object -First 1)
        $logFallo = ''
        if ($runId) { $logFallo = (@(& gh run view $runId --repo $Slug --log-failed 2>$null) -join "`n") }
        if ($logFallo -match 'espejo del corpus DWG' -and $logFallo -match 'exit code 128') {
            Log 'Fallo de infraestructura del CI: no pudo clonar el repo hermano del corpus DWG (privado). No es de MiMo y no le aviso. Reintento en 3 h; lo resuelve Sergio dando acceso al CI.'
            $script:ProximaIntegracion = (Get-Date).AddHours(3)
            return
        }
        Guardar-Fallo-CI $url $res
        $script:MensajeAviso = $MsgFalloCI + $MsgContinuar
        $script:PorAvisar = $true
        $script:AvisoDesde = Get-Date
        $script:ProximaIntegracion = (Get-Date).AddMinutes(45)
        # Cuantos commits habia cuando el CI fallo. Si MiMo commitea algo nuevo, su arreglo ya
        # esta en el arbol y no hace falta agotar los 45 min: lo mira `Paso`.
        $script:CommitsAlFallar = (Commits-Sobre-Main)
        return
    }
    $script:ProximaIntegracion = (Get-Date).AddMinutes(5)
}

function Reubicar {
    if (-not (Arbol-Limpio)) {
        if (-not $script:AvisoReubicar) {
            Log 'Mergeada: espero a que MiMo deje el arbol limpio para mover su trabajo nuevo encima de main.'
            $script:AvisoReubicar = $true
            $script:ReubicarDesde = Get-Date
        }
        # Tope: si el arbol sigue sucio 20 min despues del merge, se deja de esperar y se delega en
        # Integrar, que pausa a MiMo y rebasa el mismo. Sin esto, un fichero temporal de MiMo para
        # el ciclo entero y no se mergea nada mas.
        if ($script:ReubicarDesde -and ((Get-Date) - $script:ReubicarDesde).TotalMinutes -ge 20) {
            Log 'Arbol sucio 20 min despues del merge: dejo de esperar; lo resuelve la integracion normal.'
            $script:PorReubicar = $false
            $script:AvisoReubicar = $false
            $script:ReubicarDesde = $null
            $script:ProximaIntegracion = Get-Date
        }
        return $false
    }
    $script:AvisoReubicar = $false
    Detener-Mimo
    if (-not (Arbol-Limpio)) { $script:Mensaje = $MsgContinuar; return $true }
    & git -C $Repo fetch -q --prune origin 2>&1 | Out-Null
    $nuevos = [int](& git -C $Repo rev-list --count ($script:ShaSubido + '..' + $Rama) 2>$null | Select-Object -First 1)
    $script:PorReubicar = $false
    if (Rebasar-Sobre-Main $script:ShaSubido) {
        Remove-Item -LiteralPath $CiFallo -ErrorAction SilentlyContinue
        Log ('Integrado en main. Rama movida sobre ' + (& git -C $Repo rev-parse --short HEAD 2>$null | Select-Object -First 1) + ' con ' + $nuevos + ' commits nuevos de MiMo encima.')
        $script:Mensaje = 'Tu trabajo anterior ya esta mergeado en main y tus commits nuevos se movieron encima de main. ' + $MsgContinuar
    } else {
        Log 'Mergeada, pero no pude mover los commits nuevos encima de main. La rama queda como estaba; queda para Sergio.'
        $script:Mensaje = $MsgContinuar
    }
    $script:ProximaIntegracion = (Get-Date).AddMinutes($MinutosTrasMerge)
    return $true
}

function Avisar {
    if (-not (Arbol-Limpio) -and ((Get-Date) - $script:AvisoDesde).TotalMinutes -lt 20) { return $false }
    Log 'Relanzo a MiMo con un aviso antes de seguir con la cola.'
    Detener-Mimo
    Revertir-Dependencias
    $script:PorAvisar = $false
    $script:Mensaje = $script:MensajeAviso
    return $true
}

function Reorientar-Mimo {
    Log 'Reorientar: relanzo a MiMo en la misma sesion para que lea la orden nueva del prompt.'
    Detener-Mimo
    Revertir-Dependencias
    $script:PorReorientar = $false
    $script:Mensaje = $MsgReorientar
    return $true
}

function Paso {
    if ($script:PorReorientar) {
        $r = Reorientar-Mimo
        if (-not $script:PRUrl -and (Get-Date) -ge $script:ProximaIntegracion) { [void](Integrar) }
        return $r
    }
    if ($script:PRUrl) { Revisar-PR }
    if ($script:PorReubicar) { return (Reubicar) }
    if ($script:PorAvisar) { return (Avisar) }
    # Tras un fallo de CI se esperan 45 min para que MiMo lo arregle antes de volver a subir. Pero
    # si YA commiteo algo nuevo, esperar el resto no sirve de nada: el arreglo esta en el arbol y
    # los candados locales diran si vale. Se exige un minimo de 10 min desde el aviso para no
    # empujar a cada commit suelto (17-sep 19:30).
    if (-not $script:PRUrl -and $script:CommitsAlFallar -gt 0 -and (Get-Date) -lt $script:ProximaIntegracion) {
        if (((Get-Date) - $script:AvisoDesde).TotalMinutes -ge 10) {
            $ahora = (Commits-Sobre-Main)
            if ($ahora -gt $script:CommitsAlFallar) {
                Log ('MiMo commiteo ' + ($ahora - $script:CommitsAlFallar) + ' commit(s) desde el aviso del CI: integro ya, sin agotar los 45 min.')
                $script:CommitsAlFallar = 0
                return (Integrar)
            }
        }
    }
    if (-not $script:PRUrl -and (Get-Date) -ge $script:ProximaIntegracion) { $script:CommitsAlFallar = 0; return (Integrar) }
    return $false
}

$script:PRUrl = $null

$script:Mensaje = $null

# Si ya hay una PR abierta de la rama (por ejemplo, tras relanzar el script), la sigo si contiene main;
# si no, la actualizo en cuanto se pueda.
# Sin comillas dobles dentro de --jq: PowerShell 5.1 las borra al pasar argumentos a un ejecutable nativo,
# y con '.url + " " + .headRefOid' gh fallaba en silencio (la PR abierta nunca se seguia al arrancar).
$abiertaUrl = [string](& gh pr list --repo $Slug --head $Rama --state open --json url --jq '.[0].url' 2>$null | Select-Object -First 1)
$abiertaSha = [string](& gh pr list --repo $Slug --head $Rama --state open --json headRefOid --jq '.[0].headRefOid' 2>$null | Select-Object -First 1)
$abierta = $abiertaUrl.Trim() + ' ' + $abiertaSha.Trim()
if ($abierta -like 'https://* *') {
    $partes = $abierta.Split(' ')
    if (Contiene-Main $partes[1]) {
        $script:PRUrl = $partes[0]
        $script:ShaSubido = $partes[1]
        & gh pr merge $script:PRUrl --repo $Slug --auto --squash 2>&1 | Out-Null
        Log ('Sigo la PR abierta ' + $script:PRUrl + ' (' + $script:ShaSubido.Substring(0, 8) + '), con auto-merge.')
    } else {
        Log ('La PR abierta ' + $partes[0] + ' no contiene main al dia: la actualizo en la primera integracion (en ' + $MinutosPrimeraIntegracion + ' min).')
    }
}

# Si MiMo no esta corriendo, integro antes de relanzarlo: el arbol esta quieto.
if (-not (Get-Process -Name mimo -ErrorAction SilentlyContinue)) {
    if ($script:PorReorientar) { $script:PorReorientar = $false; $script:Mensaje = $MsgReorientar; Revertir-Dependencias }
    [void](Paso)
}

$fallos = 0
$cortas = 0

for ($v = 1; $v -le $MaxVueltas; $v++) {
    $p = $null
    $adoptado = $false
    $pausado = $false
    if ($v -eq 1 -and $Adoptar) {
        $p = Get-Process -Name mimo -ErrorAction SilentlyContinue | Select-Object -First 1
        if ($p) { $adoptado = $true; Log ('Adopto la vuelta de MiMo que ya estaba en marcha (PID ' + $p.Id + ').') }
    }
    if (-not $p) {
        $nueva = ($v -eq 1 -and -not $Adoptar)
        $m = $MsgContinuar
        if ($nueva) { $m = 'Lee el fichero .mimocode/prompt-noche.md y ejecuta esa mision sin detenerte.' }
        if ($script:PorReorientar) { $m = $MsgReorientar; $script:PorReorientar = $false }
        if ($script:Mensaje) { $m = $script:Mensaje }
        $script:Mensaje = $null
        # La reorientacion arranca una sesion nueva: con -c, la sesion de miles de pasos seguia su cola vieja
        # aunque el mensaje le pidiera releer el prompt (noche del 15-sep).
        if ($m -eq $MsgReorientar) { $nueva = $true }
        $p = Lanzar-Vuelta $v $m $nueva
    }
    $inicio = Get-Date

    while (-not $p.HasExited) {
        # Primera comprobacion enseguida si hay que reorientar; despues, cada minuto.
        if ($script:PorReorientar) { Start-Sleep -Seconds 2 } else { Start-Sleep -Seconds 60 }
        $quieto = Minutos-Sin-Actividad
        if ($quieto -ge $MinutosSinActividad) {
            Log ('Vigilante: la sesion lleva ' + $quieto + ' min sin actividad. Reinicio a MiMo.')
            Detener-Mimo
            $pausado = $true
            break
        }
        if ((Paso) -eq $true) { $pausado = $true; break }
    }

    $minutos = [math]::Round(((Get-Date) - $inicio).TotalMinutes, 1)
    $rc = 0
    if (-not $adoptado -and -not $pausado) { try { $p.WaitForExit(); $rc = $p.ExitCode } catch { $rc = 0 } }
    Log ('Vuelta ' + $v + ' termino (codigo ' + $rc + ', ' + $minutos + ' min). Commits sobre main: ' + (Commits-Sobre-Main))

    if ($pausado) { $fallos = 0; continue }

    $motivo = $null
    if (-not $adoptado) {
        # Solo stderr y los eventos de error del JSON. La salida normal trae marcas de tiempo (1789460027429) y
        # lineas de codigo donde '429' o 'quota' aparecen sin ser un aviso: la noche del 15-sep eso metio
        # esperas falsas de 5+15+30+60 min con MiMo parado.
        $cuota = 'Insufficient|quota exceeded|quota exhausted|Too Many Requests|rate.?limit|free API service|HTTP 429|"status":\s*429'
        $motivo = Select-String -LiteralPath (Join-Path $Dir ('vuelta-' + $v + '.err.txt')) -Pattern $cuota -List -ErrorAction SilentlyContinue | Select-Object -First 1
        if (-not $motivo) {
            $motivo = Select-String -LiteralPath (Join-Path $Dir ('vuelta-' + $v + '.jsonl')) -Pattern '"type":\s*"error"' -ErrorAction SilentlyContinue |
                Where-Object { $_.Line -match $cuota } | Select-Object -First 1
        }
        if ($motivo) { Log ('  Aviso del servicio: ' + $motivo.Line.Substring(0, [math]::Min(200, $motivo.Line.Length))) }
    }
    if ($rc -ne 0 -or $motivo) {
        $fallos++
        if ($fallos -ge $MaxFallosSeguidos) {
            # Para trabajar dias seguidos no se rinde: espera dos horas (cuota, red o servicio caidos) y vuelve a empezar.
            Log ($MaxFallosSeguidos.ToString() + ' vueltas seguidas con error (posible cuota, red o servicio caidos). Espero 2 horas y reintento, SIN dejar de integrar.')
            # La espera no puede ser ciega: durante ella se sigue integrando y vigilando la PR, o el
            # trabajo ya commiteado de MiMo se queda horas sin subir y sin mergear (16-sep).
            $hasta = (Get-Date).AddSeconds(7200)
            while ((Get-Date) -lt $hasta) { [void](Paso); Start-Sleep -Seconds 60 }
            $fallos = 0
            continue
        }
        $espera = @(5, 15, 30, 60, 60, 60, 60, 60)[[math]::Min($fallos - 1, 7)]
        Log ('Espero ' + $espera + ' minutos antes de reintentar, sin dejar de integrar.')
        $hasta = (Get-Date).AddMinutes($espera)
        while ((Get-Date) -lt $hasta) { [void](Paso); Start-Sleep -Seconds 60 }
        continue
    }
    $fallos = 0

    [void](Paso)
    if ($minutos -lt 3) { $cortas++ } else { $cortas = 0 }
    if ($cortas -ge 5) {
        Log 'Cinco vueltas seguidas de menos de 3 minutos: MiMo no encuentra trabajo. Espero 30 minutos.'
        Start-Sleep -Seconds 1800
        $cortas = 0
    }
}

Log 'Fin de la noche.'
Salir 0
